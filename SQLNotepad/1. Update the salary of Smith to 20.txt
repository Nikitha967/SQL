1. Update the salary of Smith to 2000
UPDATE EMP
SET SALARY = 2000
WHERE NAME = 'Smith';
---------------------------------------
2. Update the salary of King to twice his salary

UPDATE EMP
SET SALARY = SALARY * 2
WHERE NAME = 'King';
---------------------------------------
3. Construct a STUDENT table

CREATE TABLE STUDENT (
    STUDENT_ID NUMBER(10) PRIMARY KEY,
    STUDENT_NAME VARCHAR(100),
    BRANCH VARCHAR(50),
    PERCENTAGE FLOAT(7,5),
    NO_OF_BACKLOGS NUMBER
);
---------------------------------------
4. Remove Adams from the EMP table
DELETE FROM EMP
WHERE NAME = 'Adams';
---------------------------------------
5. Create a LIBRARY table
CREATE TABLE LIBRARY (
    BOOK_NO NUMBER(10) PRIMARY KEY,
    BOOK_NAME VARCHAR(100),
    AUTHOR VARCHAR(100),
    PUBLISHED_ON DATE
);
---------------------------------------
