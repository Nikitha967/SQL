package StyleProgramming;



public class StudentValidation {

	private static int[] studentIds=new int[100];

	private static int index=0;

	public static boolean nameValidation(String studentName) {

		int condition1=0;

		int condition2=0;

		/*condition to check length of name */

		int count=0;

		

		for(int i=0;i<=studentName.length()-1;i++) {

			count++;

		}

		if(count==4) {

			condition1=1;

		}

		/to check first char must be capital/

		char firstLetter=studentName.charAt(0);

		if(Character.isUpperCase(firstLetter)) {

			condition2=1;

		}

		/climax condition/

		if(condition2==1&&condition1==1) {

			return true;

		}

		else {

			return false;

		}

	}

	public static boolean studentIdValidation(int studentId) {

		int condition1=0;

		int condition2=0;

		int count=0;

		int temp=studentId;

		while(temp>0) {

			count++;

			temp/=10;

		}

		if(count==4)

			condition1=1;

		/condition 2 unique/

		boolean isUnique=true;

		for(int i=0;i<=studentIds.length-1;i++) {

			if(studentIds[i]==studentId) {

				isUnique=false;

				break;

			}

			

		}

		studentIds[index++]=studentId;

		if(isUnique) {

			condition2=1;

		}

		/climax condition/

		if(condition1==1&&condition2==1)

			return true;

		else

			return false;

	}

	public static boolean studentPercentageValidation(double studentPerc) {

		if(studentPerc>=1&&studentPerc<=99) {

			return true;

		}

		else {

			return false;

		}

	}

}
