CREATE TABLE STUDENTS
(
STUDENT ID NUMBER(10) PRIMARY KEY,
LAST NAME VARCHAR(20) NOT NULL,
FIRST NAME VARCHAR(20) NOT NULL
);

CREATE TABLE CLASSES
(
CLASS ID NUMBER(10) PRIMARY KEY,
TITLE VARCHAR(20) UNIQUE NOT NULL ,
DESCRIPTION VARCHAR(30) NOT NULL
);

CREATE TABLE ENROLLMENTS
(
ENROLLMENT ID NUMBER(10) PRIMARY KEY,
STUDENT ID NUMBER(10)
CONSTRAINT SKSK FOREIGN KEY (STUDENT ID)
REFERENCES STUDENTS(STUDENT ID),
CLASS ID NUMBER(10)
CONSTRAINT CKCK FOREIGN KEY (CLASS ID)
REFERENCES CLASSES(CLASS ID)
);

----------------------------------------------

CREATE TABLE GENRES
(
GENRE_ID NUMBER(10) PRIMARY KEY,
GENRE_NAME VARCHAR(20) NOT NULL
);

CREATE TABLE AUTHORS
(
AUTHOR_ID NUMBER(10) PRIMARY KEY,
AUTHOR_NAME VARCHAR(20) NOT NULL,
PHO_NO NUMBER(10) UNIQUE NOT NULL CHECK(LENGTH(PHO_NO)=10)
);

CREATE TABLE USERS
(
USER_ID NUMBER(10) PRIMARY KEY,
USERNAME VARCHAR(20) NOT NULL,
EMAIL VARCHAR(20) UNIQUE NOT NULL,
PHO_NO NUMBER(10) UNIQUE NOT NULL CHECK(LENGTH(PHO_NO)=10)
);

CREATE TABLE BOOKS
(
BOOK_ID NUMBER(10) PRIMARY KEY,
TITLE VARCHAR(20) UNIQUE NOT NULL,
AUTHOR_ID NUMBER(10)
CONSTRAINT AKAK FOREIGN KEY (AUTHOR_ID)
REFERENCES AUTHORS(AUTHOR_ID),
GENRE_ID NUMBER(10),
CONSTRAINT GKGK FOREIGN KEY (GENRE_ID)
REFERENCES GENRES(GENRE_ID)
);

CREATE TABLE LOANS
(
LOAN_ID NUMBER(10) PRIMARY KEY,
USER_ID NUMBER(10)
CONSTRAINT UKUK FOREIGN KEY (USER_ID)
REFERENCES USERS(USER_ID),
BOOK_ID NUMBER(10)
CONSTRAINT BKBK FOREIGN KEY (BOOK_ID)
REFERENCES BOOKS(BOOK_ID),
LOAN_DATE DATE NOT NULL,
RETURN_DATE DATE NOT NULL
);
