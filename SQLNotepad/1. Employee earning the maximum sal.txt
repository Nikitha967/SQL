1. Employee earning the maximum salary
SELECT ENAME, SAL 
FROM EMP 
WHERE SAL = (SELECT MAX(SAL)
              FROM EMP);

2.Employee earning the minimum salary

SELECT ENAME, SAL 
FROM EMP 
WHERE SAL = (SELECT MIN(SAL)
             FROM EMP);

5.Employee who earns the minimum commission
SELECT ENAME, COMM 
FROM EMP 
WHERE COMM = (SELECT MIN(COMM)
              FROM EMP
              WHERE COMM IS NOT NULL);

9.Employees earning the least annual salary
SELECT  (SAL * 12) AS ANNUAL_SALARY 
FROM EMP 
WHERE SAL = (SELECT MIN(SAL)

10. Employees whose annual salary is more than all Salesmen


SELECT ENAME, SAL * 12 AS ANNUAL_SALARY 
FROM EMP 
WHERE (SAL * 12) > (SELECT MAX(SAL * 12)
