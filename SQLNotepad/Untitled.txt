public class InvertedTriangle {
    public static void main(String[] args) {
        int n = 5; 
        for (int i = n; i >= 1; i--) {
            for (int j = 1; j <= i; j++) {
                System.out.print("* ");
            }
            System.out.println();
        }
    }
}

public class RightAlignedInvertedTriangle {
    public static void main(String[] args) {
        int n = 5; // Number of rows
        for (int i = n; i >= 1; i--) {
            for (int j = 0; j < n - i; j++) {
                System.out.print(" "); // Print spaces
            }
            for (int j = 0; j < i; j++) {
                System.out.print("*"); // Print stars
            }
            System.out.println();
        }
    }
}
